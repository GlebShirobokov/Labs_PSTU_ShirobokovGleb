#include <iostream>

using namespace std;

int main() {
	int a[5] = { 5, 3, 4, 2, 1 };
	for (int i = 1; i < 5; ++i) {
		bool f = false;
		for (int j = 0; j < i && !f; ++j) {
			if (a[j] > a[i]) {
				int tmp = a[i];
				for (int h = i; h > j; --h) a[h] = a[h - 1];
				a[j] = tmp;
				f = true;
			}
		}
	}
	for (int i = 0; i < 5; ++i) cout << a[i] << ' ';
	cout << endl;

	int a1[5] = { 5, 3, 4, 2, 1 };
	for (int i = 0; i < 4; ++i) {
		int min = a1[i];
		for (int j = i + 1; j < 5; ++j) {
			if (min > a1[j]) {
				min = a1[j];
				a1[j] = a1[i];
				a1[i] = min;
			}
		}
	}
	for (int i = 0; i < 5; ++i) cout << a1[i] << ' ';
	cout << endl;

	int a2[5] = { 5, 3, 4, 2, 1 };
	for (int i = 0; i < 5; ++i) {
		for (int j = i + 1; j < 5; ++j) {
			if (a2[i] > a2[j]) {
				int tmp = a2[j];
				a2[j] = a2[i];
				a2[i] = tmp;
			}
		}
	}
	for (int i = 0; i < 5; ++i) cout << a2[i] << ' ';
	cout << endl;
}
