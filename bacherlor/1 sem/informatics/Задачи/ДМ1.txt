
#include <iostream>


using namespace std;

int main() {
	int n;
	cin >> n;
	int* a = new int[n];
	
		for (int i = 0; i < n; ++i) cin >> a[i];
		for (int i = 0; i < n; ++i) {
			if (a[i] % 2 == 0) a[i] = a[i - 1] + a[i + 1];
			if (i == 0) a[i] = 0;
			if (i == n - 1) a[i] = 0;

		}
		for (int i = 0; i < n; ++i) cout << a[i] << ' ';
		delete[] a;
		a = nullptr;
		return 0;
	
}
